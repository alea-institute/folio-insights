    ```python
    """folio-insights Dagger build pipeline.

    D-10: full CI (build + lint + test + publish + deploy-trigger).
    D-08: bit-identical digest via SOURCE_DATE_EPOCH + `--require-hashes`.

    Stage ordering (Claude's discretion per CONTEXT.md line 64):
      Parallel: build-web | build-worker | lint
      Serial:   test (needs python runtime image from build-web)
      Serial:   publish (after all above)
      Serial:   deploy (only on $CI + main branch)

    Invoke: `python -m dagger.build [--no-deploy] [--tag <tag>]`
    """
    from __future__ import annotations

    import argparse
    import asyncio
    import os
    import subprocess
    import sys
    from pathlib import Path

    import dagger


    # ---------------------------------------------------------------------------
    # Constants — load base digests from .env.docker if present, else .env.docker.example
    # ---------------------------------------------------------------------------
    REPO_ROOT = Path(__file__).resolve().parents[1]


    def _load_digests() -> dict[str, str]:
        """Read `.env.docker` (or `.env.docker.example`) into a dict."""
        for candidate in (".env.docker", ".env.docker.example"):
            path = REPO_ROOT / candidate
            if path.exists():
                out: dict[str, str] = {}
                for line in path.read_text().splitlines():
                    line = line.strip()
                    if not line or line.startswith("#") or "=" not in line:
                        continue
                    k, v = line.split("=", 1)
                    out[k] = v.strip()
                return out
        raise FileNotFoundError("Neither .env.docker nor .env.docker.example found")


    def _source_date_epoch() -> str:
        """Git HEAD commit timestamp → SOURCE_DATE_EPOCH (Gate 5 step 2)."""
        return subprocess.check_output(
            ["git", "log", "-1", "--pretty=%ct"], cwd=REPO_ROOT
        ).decode().strip()


    # Gate 5 step 10: explicit exclude list (defence-in-depth beyond .dockerignore)
    BUILD_CTX_EXCLUDE = [
        ".git",
        ".github",
        ".planning",
        ".claude",
        "output",
        "fixtures/bench.nq",
        "fixtures/bench-*.nq",
        "node_modules",
        "viewer/node_modules",
        "viewer/.svelte-kit",
        "viewer/build",
        "**/__pycache__",
        "**/*.pyc",
        ".venv",
        "dagger/.venv",
        "dagger/__pycache__",
    ]


    async def _build_image(
        client: dagger.Client,
        *,
        dockerfile: str,
        tag: str,
        sde: str,
        digests: dict[str, str],
    ) -> tuple[str, str]:
        """Build an image via Dagger's dockerfile integration. Returns (tag, digest)."""
        src = client.host().directory(str(REPO_ROOT), exclude=BUILD_CTX_EXCLUDE)

        # Dagger's `Directory.docker_build` respects ARG SOURCE_DATE_EPOCH (Pitfall 4 — BOTH env+arg)
        container = src.docker_build(
            dockerfile=dockerfile,
            build_args=[
                dagger.BuildArg(name="SOURCE_DATE_EPOCH", value=sde),
            ],
        )
        # Belt-and-braces: also set the env var on the resulting container
        container = container.with_env_variable("SOURCE_DATE_EPOCH", sde)

        digest = await container.publish(tag)
        return tag, digest


    async def _lint(client: dagger.Client, sde: str) -> None:
        """Run ruff against src/ api/ tests/ — D-10 lint stage."""
        src = client.host().directory(str(REPO_ROOT), exclude=BUILD_CTX_EXCLUDE)
        result = (
            client.container()
            .from_("python:3.11-slim")  # ephemeral — lint does not ship to Gate 5
            .with_env_variable("SOURCE_DATE_EPOCH", sde)
            .with_directory("/app", src)
            .with_workdir("/app")
            .with_exec(["pip", "install", "--no-cache-dir", "ruff>=0.6.0"])
            .with_exec(["ruff", "check", "src/", "api/", "tests/", "dagger/"])
        )
        await result.sync()


    async def _test(client: dagger.Client, sde: str) -> None:
        """Run pytest quick-suite (excluding benchmarks — Gates 2/3/4 run separately in Plan 07)."""
        src = client.host().directory(str(REPO_ROOT), exclude=BUILD_CTX_EXCLUDE)
        result = (
            client.container()
            .from_("python:3.11-slim")
            .with_env_variable("SOURCE_DATE_EPOCH", sde)
            .with_directory("/app", src)
            .with_workdir("/app")
            .with_exec(["pip", "install", "--no-cache-dir", "--require-hashes", "-r", "requirements.dev.lock"])
            .with_exec(["pytest", "-x", "--ff", "-q", "--benchmark-skip"])
        )
        await result.sync()


    async def main(args: argparse.Namespace) -> None:
        """Pipeline driver — build + lint + test in parallel-where-possible, then publish."""
        sde = _source_date_epoch()
        digests = _load_digests()
        tag_suffix = args.tag or sde

        async with dagger.Connection(dagger.Config(log_output=sys.stderr)) as client:
            # Parallel stages (Gate 5 parallelizability)
            web_task = _build_image(
                client,
                dockerfile="Dockerfile.web",
                tag=f"ttl.sh/fi-web:{tag_suffix}",
                sde=sde,
                digests=digests,
            )
            worker_task = _build_image(
                client,
                dockerfile="Dockerfile.worker",
                tag=f"ttl.sh/fi-worker:{tag_suffix}",
                sde=sde,
                digests=digests,
            )
            lint_task = _lint(client, sde)

            # Test depends on nothing Dagger-wise but sequenced after for log readability
            results = await asyncio.gather(web_task, worker_task, lint_task)
            web_result, worker_result, _ = results

            await _test(client, sde)

            print(f"SOURCE_DATE_EPOCH={sde}")
            print(f"WEB: {web_result[0]} @ {web_result[1]}")
            print(f"WORKER: {worker_result[0]} @ {worker_result[1]}")

        # Deploy (serial, post-success) — skipped on --no-deploy or absent RAILWAY_TOKEN
        if args.no_deploy or not os.environ.get("RAILWAY_TOKEN"):
            print("Skipping Railway deploy (--no-deploy or RAILWAY_TOKEN missing)")
            return
        from dagger.railway import deploy_service
        deploy_service("web", image=web_result[0])
        deploy_service("worker", image=worker_result[0])
    ```

    ```python
    """Railway deploy wrapper — invoked by dagger/build.py after successful publish.

    D-10: deploy stage. Uses Railway CLI via subprocess (simpler than GraphQL token dance).
    Requires RAILWAY_TOKEN env var.
    """

    import logging
    import os
    import subprocess

    def deploy_service(service_name: str, *, image: str | None = None) -> None:
        """Trigger Railway deploy for one service.

        Args:
            service_name: "web" or "worker" — must match railway.toml service names.
            image: Optional pre-built image ref (registry URL); if None, Railway rebuilds.
        """
        token = os.environ.get("RAILWAY_TOKEN")
        if not token:
            raise RuntimeError("RAILWAY_TOKEN not set — cannot deploy")

        cmd = ["railway", "up", "--service", service_name, "--ci"]
        if image:
            # Railway's image-deploy path; alternatively use `railway deploy` with pre-pushed image
            cmd.extend(["--image", image])

        logger.info("Deploying %s → %s", service_name, image or "<Railway-rebuild>")
        result = subprocess.run(

            env={**os.environ, "RAILWAY_TOKEN": token},
            capture_output=True,
            text=True,
            check=False,
        )
        if result.returncode != 0:
            logger.error("Railway deploy failed: %s", result.stderr)
            raise RuntimeError(f"railway up failed for {service_name}: {result.stderr}")
        logger.info("Deployed %s", service_name)


    if __name__ == "__main__":
        # Allow: `python -m dagger.railway web` for manual re-deploys
        import sys
        deploy_service(sys.argv[1])
    ```

    ```python
    """Gate 5 — bit-identical digest (REQ-OBS-04, D-08).

    Two assertion modes:
      (1) LOCAL: two back-to-back Dagger builds with same SOURCE_DATE_EPOCH → identical digests.
          This proves the local pipeline is deterministic. Mandatory (always runs in CI).
      (2) RAILWAY: deployed image digest (pulled from Railway registry) == locally-built digest.
          This proves the end-to-end deploy preserves reproducibility. Skipped when RAILWAY_TOKEN absent.
    """
    from __future__ import annotations

    import os
    import subprocess

    import pytest


    def _inspect_digest(image_ref: str) -> str | None:
        """Run `docker inspect --format '{{.Id}}'`. Returns None if image missing locally."""
        result = subprocess.run(
            ["docker", "inspect", "--format", "{{.Id}}", image_ref],
            capture_output=True, text=True, check=False,
        )
        if result.returncode != 0:
            return None
        return result.stdout.strip()


    def _dagger_build(tag: str) -> str:
        """Invoke `python -m dagger.build --no-deploy --tag <tag>` and parse digest from stdout."""
        result = subprocess.run(
            ["python", "-m", "dagger.build", "--no-deploy", "--tag", tag],
            capture_output=True, text=True, check=True, timeout=600,
        )
        # Expected stdout line: `WEB: ttl.sh/fi-web:<tag> @ ttl.sh/fi-web:<tag>@sha256:<digest>`
        for line in result.stdout.splitlines():
            if line.startswith("WEB:"):
                parts = line.split("@")
                if len(parts) >= 3:
                    return "sha256:" + parts[-1].split(":")[-1]
        raise AssertionError(f"Could not parse digest from dagger.build output:\n{result.stdout}")


    @pytest.mark.gate5
    @pytest.mark.slow
    def test_local_dagger_builds_bit_identical() -> None:
        """Mode 1 (always runs): two local Dagger builds must produce identical digests."""
    ```
