    ```python
    """HermiT reasoning harness (D-11 full 1M ABox).

    RISK-1 mitigation: runs in worker tier only (JVM bloat excluded from web per REQ-QUALITY-03).
    Pitfall 3: first invocation costs 5-15s JVM cold-start — measured separately.

    Architectural-responsibility note: this harness is a THIN WRAPPER around
    `owlready2.sync_reasoner_hermit()`. Per RESEARCH.md §Don't Hand-Roll line 476,
    we do NOT shell out to `java -jar HermiT.jar` directly — owlready2 handles
    tempfile, classpath, error parsing.
    """
    from __future__ import annotations

    import logging
    import time
    from dataclasses import dataclass, field
    from pathlib import Path
    from typing import Any

    import owlready2

    logger = logging.getLogger(__name__)


    @dataclass
    class HermitResult:
        """Structured output of a single HermiT run."""
        ontology: str
        xmx_mb: int
        elapsed_s: float
        consistent: bool
        inconsistent_classes: list[str] = field(default_factory=list)

        def as_dict(self) -> dict[str, Any]:
            return {
                "ontology": self.ontology,
                "xmx_mb": self.xmx_mb,
                "elapsed_s": self.elapsed_s,
                "consistent": self.consistent,
                "inconsistent_classes": self.inconsistent_classes,
            }


    class HermitHarness:
        """D-11 harness: full-corpus HermiT reasoning with Xmx tuning.

        Usage:
            harness = HermitHarness(xmx_mb=4096)
            result = harness.reason(Path("fixtures/bench-abox-1m.owl"))
            print(f"HermiT: {result.elapsed_s:.1f}s, consistent={result.consistent}")
        """

        def __init__(self, xmx_mb: int = 4096) -> None:
            """
            Args:
                xmx_mb: JVM max heap in MB. D-11 tuning target — start 4096, measure, adjust.
            """
            self.xmx_mb = xmx_mb
            # owlready2 reads this module-global at subprocess-spawn time (docs)
            owlready2.JAVA_MEMORY = xmx_mb

        def reason(self, ontology_path: Path) -> HermitResult:
            """Load ontology → sync_reasoner_hermit() → capture timing + consistency."""
            # Fresh world prevents cross-test state leak (owlready2 has process-global default_world)
            world = owlready2.World()
            onto = world.get_ontology(str(ontology_path)).load()

            t0 = time.perf_counter()
            try:
                with onto:
                    owlready2.sync_reasoner_hermit(
                        x=[onto],
                        infer_property_values=True,
                    )
            except Exception as exc:
                elapsed = time.perf_counter() - t0
                logger.error("HermiT failed after %.2fs: %s", elapsed, exc)
                raise
            elapsed = time.perf_counter() - t0

            inconsistent = list(onto.inconsistent_classes())
            result = HermitResult(
                ontology=str(ontology_path),
                xmx_mb=self.xmx_mb,
                elapsed_s=elapsed,
                consistent=len(inconsistent) == 0,
                inconsistent_classes=[str(c) for c in inconsistent],
            )
            logger.info(
                "HermiT reasoned %s in %.2fs (Xmx=%dMB, consistent=%s)",
                ontology_path.name, elapsed, self.xmx_mb, result.consistent,
            )
            return result
    ```

        """Run hyperfine, parse times[] from JSON, compute P95 in ms."""
        cmd = [

            "--warmup", str(warmup),
            "--runs", str(runs),
            "--export-json", "-",
            "--style", "none",
            f"curl -s -o /dev/null {url}",

        data = json.loads(result.stdout)
        times_s = data["results"][0]["times"]  # list of seconds
        times_ms = sorted(t * 1000 for t in times_s)
        # P95 = 95th percentile
        p95_ms = times_ms[int(0.95 * len(times_ms))]
        return p95_ms
