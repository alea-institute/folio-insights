    ```python
    """Gate 2 P95 SPARQL harness (QUALITY-01, D-05).

    Stateless entrypoint used by:
      - `pytest-benchmark` via `tests/bench/test_gate2_sparql.py`
      - Plan 08 verdict scripts (DECISION.md ingestion)

    D-05 discipline: every call goes through the same code path so measurements are comparable.
    """
    from __future__ import annotations

    from pathlib import Path

    from pyoxigraph import NamedNode

    from folio_insights.store import PyoxigraphStore

    # ---------------------------------------------------------------------------
    # Per-query named-graph prune table (Gate 2 tuning step 3)
    # Populated from PRD §20 + 00-02-SUMMARY.md graph-layout.
    # None = no pruning (full graph scan); named-graph list = prune scan.
    # ---------------------------------------------------------------------------
    ADVOCACY_G = NamedNode("urn:folio:advocacy")
    FRE_G = NamedNode("urn:folio:fre")
    REST_G = NamedNode("urn:folio:restatement")
    GOV_G = NamedNode("urn:folio:governance")
    TBOX_G = NamedNode("urn:folio:tbox")

    GOLD_QUERY_NAMED_GRAPHS: dict[str, list[NamedNode] | None] = {
        # Each key is the file stem (without .sparql); value is named_graphs= param for pyoxigraph Store.query
        "q01_confidence_threshold": [ADVOCACY_G],
        "q02_advocacy_with_provenance": [ADVOCACY_G],
        "q03_polysemy_siblings": [ADVOCACY_G, FRE_G, REST_G],  # cross-corpus
        "q04_supersession_chain": None,  # supersession crosses graphs; no prune
        "q05_framework_filter": [FRE_G, REST_G],
        "q06_as_of_historical": None,  # bitemporal spans all corpora
        "q07_axiom_closure": [TBOX_G, ADVOCACY_G],  # axioms in tbox, deps in advocacy
        "q08_seven_principles_conflict": [ADVOCACY_G, GOV_G],  # P1 violations recorded in governance
        "q09_author_cluster_by_framework": [ADVOCACY_G],
        "q10_governance_attestation": [GOV_G],
        "q11_cross_corpus_traversal": None,  # cross-corpus by definition
        "q12_deprecated_with_successor": [ADVOCACY_G, TBOX_G],
        "q13_confidence_histogram": [ADVOCACY_G],
    }


    def load_store_optimized(corpus_path: Path) -> PyoxigraphStore:
        """Gate 2 steps 1+2: bulk_load then optimize. Returns fully-loaded store.

        This is the SLOW setup path — callers SHOULD cache the result across queries.
        Expected one-time cost: 10-60s for 1M triples (RocksDB bulk ingest + compaction).
        """
        store = PyoxigraphStore(path=None)  # in-memory; Phase 13 switches to path= for persistence
        store.bulk_load_nquads(corpus_path)
        store.optimize()
        return store


    def load_and_query(
        corpus_path: Path,
        query_path: Path,
        named_graphs: list[NamedNode] | None | str = "auto",
    ) -> list:
        """Gate 2 measurement entrypoint — bulk_load → optimize → query.

        Args:
            corpus_path: Path to `fixtures/bench.nq` (or similar).
            query_path: Path to a `.sparql` file in `fixtures/gold_queries/`.
            named_graphs:
                - "auto" (default): look up in GOLD_QUERY_NAMED_GRAPHS by file stem
                - None: force full graph scan (no pruning)
                - list[NamedNode]: explicit prune list

        Returns:
            List of query result rows.

        NOTE: This function rebuilds the store on every call. For pytest-benchmark
              pedantic mode with rounds=20, this means 20 bulk-loads. That's expensive
              but INTENTIONAL — D-05 measures end-to-end, NOT warm-store query time.
              If you want warm-store P95, use `load_store_optimized()` once + call
              `store.query_rdf12()` directly in the benchmark closure.
        """
        store = load_store_optimized(corpus_path)

        if named_graphs == "auto":
            named_graphs = GOLD_QUERY_NAMED_GRAPHS.get(query_path.stem)

        sparql = query_path.read_text()
        return store.query_rdf12(sparql, named_graphs=named_graphs)


    def query_only(
        store: PyoxigraphStore,
        query_path: Path,
        named_graphs: list[NamedNode] | None | str = "auto",
    ) -> list:
        """Warm-path variant — query against a pre-loaded store. Use this for
        repeated-query P95 (Gate 2 playbook step 4: warm cache).
        """
        if named_graphs == "auto":
            named_graphs = GOLD_QUERY_NAMED_GRAPHS.get(query_path.stem)
        sparql = query_path.read_text()
        return store.query_rdf12(sparql, named_graphs=named_graphs)
    ```

    ```python
    """Gate 2 P95 SPARQL harness (QUALITY-01, D-05).

    D-05 rule (CONTEXT.md):
      - P95 < 500ms hard target
      - ≤800ms accept-with-SLO after 6-pass tuning
      - >800ms triggers `pivot=fuseki` verdict in DECISION.md

    This test runs the full 13-query gold-query set in pytest-benchmark pedantic mode.
    Results autosave to bench-results.json for Plan 08 DECISION.md ingestion.

    Run:
      - Baseline (untuned warm-cache):   uv run pytest tests/bench/test_gate2_sparql.py -m benchmark --benchmark-only --benchmark-autosave
      - Cold-cache variant:              uv run pytest tests/bench/test_gate2_sparql.py::test_gate2_p95_cold -m benchmark --benchmark-only
    """
    from __future__ import annotations

    from pathlib import Path

    import pytest

    from folio_insights.bench.gate2_harness import (
        load_and_query,
        load_store_optimized,
        query_only,
    )

    QUERIES_DIR = Path(__file__).resolve().parents[2] / "fixtures" / "gold_queries"
    GOLD_QUERIES = sorted(QUERIES_DIR.glob("q*.sparql"))

    # D-05 thresholds (CONTEXT.md)
    GATE2_HARD_TARGET_MS = 500
    GATE2_SLO_CEILING_MS = 800


    @pytest.fixture(scope="module")
    def warm_store(bench_1m_corpus: Path):
        """Load + optimize once per module — warm-cache reuse (Gate 2 tuning step 4)."""
        return load_store_optimized(bench_1m_corpus)


    @pytest.mark.benchmark(group="gate2-warm-p95")
    @pytest.mark.parametrize("query_file", GOLD_QUERIES, ids=lambda p: p.stem)
    def test_gate2_p95_warm(benchmark, warm_store, query_file: Path) -> None:
        """Warm-cache P95 — the authoritative Gate 2 measurement.

        Tuning steps applied:
          1. bulk_load via `Store.bulk_load()` (Plan 03's wrapper)
          2. `Store.optimize()` after load (Plan 03's wrapper)
          3. `named_graphs=` prune (gate2_harness.GOLD_QUERY_NAMED_GRAPHS)
          4. Warm cache (this test — 3 warmup rounds, 20 measured rounds)
          5. SPARQL query-plan rewrites (baked into the .sparql files per Gate 1 rewrite pass)
          6. RocksDB tuning (limited leverage — in-memory for Phase 0; Phase 13 opens persistent)
        """

            )

            iterations=1,
        )
    ```
