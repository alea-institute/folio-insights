  **Step 3 — Create `src/folio_insights/bench/generator.py`:**

  Follow PATTERNS.md §Plan P0-B #8 skeleton + §Shared 3 determinism rules + §Shared 5 Python idioms.

  ```python
  """1M-triple scaled-real corpus generator (D-13, D-14, D-15, D-16).

  Scope:
  - Re-extract v1 advocacy + FRE + Restatement under v2 pipeline (D-01, D-13)
  - Replay bitemporal edits/supersessions to reach --target count
  - Emit N-Quads in THREE named graphs: advocacy / fre / restatement (STORAGE-03)
  - Use RDF 1.2 annotation-pipe serialization semantics — NEVER `<<>>` subject
    position (Pitfall 1)
  - Single `random.Random(seed)` instance threaded everywhere (Pitfall 7)

  Determinism contract (D-15):
  - Same --seed produces byte-identical output across runs.
  - sort() every collection before iteration.
  - Never import numpy.random; never call module-level random.choice.
  """
  from __future__ import annotations

  import hashlib
  import logging
  import random
  from dataclasses import dataclass, field
  from pathlib import Path

  from pyoxigraph import (
      NamedNode,
      Literal,
      Quad,
      RdfFormat,
      Store,
      Triple,
  )

  from folio_insights.bench.profiles import PROFILES, PhaseProfile

  logger = logging.getLogger(__name__)

  # Vocab namespaces — MUST match Phase 8 fi:* when those land; for Phase 0
  # the generator uses stable canonical IRIs.
  FI = "https://folio-insights.aleainstitute.ai/vocab/"
  CORPUS = "https://folio-insights.aleainstitute.ai/corpus/"
  SHARD = "https://folio-insights.aleainstitute.ai/shard/"
  CONCEPT = "https://folio-insights.aleainstitute.ai/concept/"


  @dataclass
  class BenchGenerator:
      """D-13 scaled-real generator.

      Phase 0 implementation emits structurally-correct RDF-12 data derived from
      a seeded RNG — does NOT invoke the v1 PipelineOrchestrator live (avoids
      bootstrap circularity; v1 corpora are mined for ratio data only via
      PROFILES).

      When Phase 10 lands the real Stage 8 Shard Minter, this class gains an
      optional `real_extraction=True` path that routes through v1 extraction.
      For Gate 1-5 measurement, seeded synthesis of the structural pattern is
      sufficient — the gates test syntax + perf + image + digest, not
      extraction fidelity.
      """
      seed: int
      profile_name: str = "phase-0-gate"
      _rng: random.Random = field(init=False)
      _profile: PhaseProfile = field(init=False)

      def __post_init__(self) -> None:
          # Pitfall 7: single explicit RNG instance, threaded everywhere below.
          self._rng = random.Random(self.seed)
          if self.profile_name not in PROFILES:
              raise ValueError(
                  f"Unknown profile {self.profile_name!r}; available: {sorted(PROFILES)}"
              )
          self._profile = PROFILES[self.profile_name]

      def generate(self, target_triples: int, output_path: Path) -> Path:
          """Produce a deterministic N-Quads file with `target_triples` quads.

          Side-effect: writes output_path (parent dirs created).
          Return: output_path (echoed for CLI).
          """
          output_path = Path(output_path)
          output_path.parent.mkdir(parents=True, exist_ok=True)

          # In-memory Store — we bulk-add, then dump. Disk-backed Store would
          # introduce RocksDB mtime variance into the N-Quads output path
          # (through file IO order), breaking determinism.
          store = Store(path=None)
          triples_emitted = 0

          # Pitfall 7: pre-sort the corpus-name list; iterate in deterministic order.
          corpora_with_shares = sorted(
              [
                  ("advocacy", self._profile.advocacy_share),
                  ("fre", self._profile.fre_share),
                  ("restatement", self._profile.restatement_share),
              ],
              key=lambda x: x[0],
          )

          for corpus_name, share in corpora_with_shares:
              graph_iri = NamedNode(f"{CORPUS}{corpus_name}")
              corpus_target = int(target_triples * share)
              emitted_for_corpus = 0
              shard_idx = 0
              while emitted_for_corpus < corpus_target:
                  subtype = self._pick_subtype()
                  # Each shard emits 6-12 quads on average (base shard + 2-3
                  # annotation quads for confidence/validFrom/extractedBy +
                  # edit_rounds × 1 supersession quad).
                  shard_quads = self._emit_shard_quads(
                      corpus_name=corpus_name,
                      subtype=subtype,
                      shard_idx=shard_idx,
                      graph_iri=graph_iri,
                  )
                  for quad in shard_quads:
                      store.add(quad)
                      emitted_for_corpus += 1
                      triples_emitted += 1
                      if emitted_for_corpus >= corpus_target:
                          break
                  shard_idx += 1

          logger.info(
              "Emitting %d quads from store to %s",
              triples_emitted,
              output_path,
          )

          # Dump as N-Quads (all named graphs in one file)
          with open(output_path, "wb") as f:
              store.dump(f, format=RdfFormat.N_QUADS)

          return output_path

      def _pick_subtype(self) -> str:
          """Weighted random subtype per D-03 ratios, using self._rng (Pitfall 7)."""
          weights = self._profile.subtype_weights
          # Pitfall 7: sort keys before iteration for determinism
          keys = sorted(weights.keys())
          cumweights = []
          total = 0.0
          for k in keys:
              total += weights[k]
              cumweights.append(total)
          r = self._rng.random() * total
          for k, cw in zip(keys, cumweights, strict=True):
              if r <= cw:
                  return k
          return keys[-1]

      def _emit_shard_quads(
          self,
          corpus_name: str,
          subtype: str,
          shard_idx: int,
          graph_iri: NamedNode,
      ) -> list[Quad]:
          """Emit one shard's worth of quads in the given named graph.

          Structure:
          - Base quads: shard a fi:<Subtype>, fi:corpus, fi:framework, fi:utrum
          - Annotation pipe SEMANTICS as reified quads: attach confidence
            + validFrom to the core subject triple (RDF-12 object-position-only
            constraint honored; we reify rather than emit `<<>>`).
          - Edit-round quads: fi:supersededBy for profile.edit_rounds - 1 edits.
          """
          shard_iri = NamedNode(f"{SHARD}{corpus_name}/{shard_idx:08x}")
          a_pred = NamedNode("http://www.w3.org/1999/02/22-rdf-syntax-ns#type")
          out: list[Quad] = []

          # Base type quad
          out.append(Quad(shard_iri, a_pred, NamedNode(f"{FI}{subtype}Shard"), graph_iri))
          # Corpus quad
          out.append(
              Quad(shard_iri, NamedNode(f"{FI}corpus"), NamedNode(f"{CORPUS}{corpus_name}"), graph_iri)
          )
          # Framework quad (deterministic per shard_idx)
          fw_choices = ["us.federal.frcp", "us.federal.fre", "us.restatement.contracts"]
          fw = fw_choices[shard_idx % len(fw_choices)]
          out.append(
              Quad(shard_iri, NamedNode(f"{FI}framework"),
                   NamedNode(f"https://folio-insights.aleainstitute.ai/framework/{fw}"),
                   graph_iri)
          )

          # Subject-concept triple with annotation-pipe semantics:
          # Emit the core triple PLUS reified annotation quads for confidence.
          # RDF-12 annotation-pipe Turtle `?s ?p ?o {| fi:confidence ?c |}` equivalence
          # in N-Quads form uses rdf:Statement reification — this is what Oxigraph emits
          # when asked to dump annotation-pipe Turtle to N-Quads (no <<...>> subject terms).
          concept = NamedNode(f"{CONCEPT}{subtype.lower()}-{shard_idx % 50}")
          subject_pred = NamedNode(f"{FI}subject")
          out.append(Quad(shard_iri, subject_pred, concept, graph_iri))

          # Annotation on the subject triple — reified (RDF-12 object-only constraint)
          stmt_iri = NamedNode(f"{SHARD}{corpus_name}/{shard_idx:08x}/stmt0")
          rdf_stmt = NamedNode("http://www.w3.org/1999/02/22-rdf-syntax-ns#Statement")
          rdf_subj = NamedNode("http://www.w3.org/1999/02/22-rdf-syntax-ns#subject")
          rdf_pred = NamedNode("http://www.w3.org/1999/02/22-rdf-syntax-ns#predicate")
          rdf_obj = NamedNode("http://www.w3.org/1999/02/22-rdf-syntax-ns#object")
          out.append(Quad(stmt_iri, a_pred, rdf_stmt, graph_iri))
          out.append(Quad(stmt_iri, rdf_subj, shard_iri, graph_iri))
          out.append(Quad(stmt_iri, rdf_pred, subject_pred, graph_iri))
          out.append(Quad(stmt_iri, rdf_obj, concept, graph_iri))
          conf = round(0.5 + self._rng.random() * 0.5, 3)
          out.append(
              Quad(stmt_iri, NamedNode(f"{FI}confidence"),
                   Literal(str(conf),
                           datatype=NamedNode("http://www.w3.org/2001/XMLSchema#decimal")),
                   graph_iri)
          )

          # Epistemic status for DisputedProposition
          if subtype == "DisputedProposition":
              out.append(
                  Quad(shard_iri, NamedNode(f"{FI}epistemicStatus"),
                       Literal("aporetic"), graph_iri)
              )

          # Bitemporal-edit rounds — supersession chain
          for round_idx in range(self._profile.edit_rounds - 1):
              succ_iri = NamedNode(
                  f"{SHARD}{corpus_name}/{shard_idx:08x}/v{round_idx + 1:02x}"
              )
              out.append(
                  Quad(shard_iri, NamedNode(f"{FI}supersededBy"), succ_iri, graph_iri)
              )

          # Adversarial-density injection (Phase 16 profile)
          if self._rng.random() < self._profile.adversarial_density:
              # Deep-graph edge for adversarial traversal test
              adv_target = NamedNode(f"{SHARD}adversarial/{self._rng.randrange(1000):08x}")
              out.append(
                  Quad(shard_iri, NamedNode(f"{FI}dependsOnAxiom"), adv_target, graph_iri)
              )

          return out
  ```

  ```python
  """Gate 0 prerequisite: bench generator determinism (D-15, D-14).

  These tests enforce Pitfall 7 discipline: --seed 42 MUST produce
  bit-identical output across runs; --seed 42 vs --seed 43 MUST differ.
  """
  from __future__ import annotations

  import hashlib
  import subprocess
  from pathlib import Path

  import pytest

  SMOKE_TARGET = 10_000  # 10k — fast test iteration (~5s); D-15 logic is scale-invariant


  def _sha256(path: Path) -> str:
      h = hashlib.sha256()
      with open(path, "rb") as f:
          for chunk in iter(lambda: f.read(1 << 16), b""):
              h.update(chunk)
      return h.hexdigest()


  def _run_gen(tmp_path: Path, seed: int, out_name: str) -> Path:
      out_path = tmp_path / out_name
      result = subprocess.run(
          [
              "uv", "run", "folio-insights", "bench", "gen",
              "--seed", str(seed),
              "--target", str(SMOKE_TARGET),
              "--out", str(out_path),
          ],
          capture_output=True, text=True, check=True,
      )
      assert out_path.exists(), f"gen didn't produce output\nstderr: {result.stderr}"
      return out_path


  def test_same_seed_produces_bit_identical_output(tmp_path: Path) -> None:
      """D-15: --seed 42 twice => byte-identical files."""
      a = _run_gen(tmp_path, seed=42, out_name="a.nq")
      b = _run_gen(tmp_path, seed=42, out_name="b.nq")
      assert _sha256(a) == _sha256(b), \
          f"Pitfall 7 violation: seeded output is not deterministic\n  a={_sha256(a)}\n  b={_sha256(b)}"


  def test_different_seed_produces_different_output(tmp_path: Path) -> None:
      """Seed actually matters (not just structural determinism)."""
      a = _run_gen(tmp_path, seed=42, out_name="a.nq")
      c = _run_gen(tmp_path, seed=43, out_name="c.nq")
      assert _sha256(a) != _sha256(c), "Seed has no effect — RNG probably not threaded"


  def test_output_contains_no_sparql_star_subject_terms(tmp_path: Path) -> None:
      """Pitfall 1 BLOCKING: banned `<<?s ?p ?o>>` subject-position terms.

      Any literal occurrence of `<<` in N-Quads output is a bug — RDF-12
      annotations must use object-position or reification only.
      """
      a = _run_gen(tmp_path, seed=42, out_name="a.nq")
      raw = a.read_bytes()
      assert b"<<" not in raw, "Generator emitted banned `<<...>>` subject-position triple term"


  def test_output_roundtrips_through_pyoxigraph_bulk_load(tmp_path: Path) -> None:
      """Generator output must be parseable by pyoxigraph.Store.bulk_load."""
      from pyoxigraph import Store, RdfFormat
      a = _run_gen(tmp_path, seed=42, out_name="a.nq")
      store = Store(path=None)
      with open(a, "rb") as f:
          store.bulk_load(f, format=RdfFormat.N_QUADS)
      store.optimize()
      # If we got here, parse + index succeeded.
      # Sanity: at least 5k quads landed (we requested 10k target; share-rounding may drop a few).
      count = sum(1 for _ in store)
      assert count >= SMOKE_TARGET - 100, f"expected ~{SMOKE_TARGET}, got {count}"


  def test_profile_flag_routes_to_different_output(tmp_path: Path) -> None:
      """D-16: --profile changes output (different ratios → different SHA)."""
      gate = tmp_path / "gate.nq"
      adv = tmp_path / "adv.nq"
      for profile, out in [("phase-0-gate", gate), ("phase-16-sparql-adversarial", adv)]:
          subprocess.run(
              ["uv", "run", "folio-insights", "bench", "gen",
               "--seed", "42", "--target", str(SMOKE_TARGET),
               "--profile", profile, "--out", str(out)],
              check=True, capture_output=True,
          )
      assert _sha256(gate) != _sha256(adv), "--profile did not change output"
  ```
